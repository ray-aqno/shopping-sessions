cmake_minimum_required(VERSION 3.20)
project(shopping_sessions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(SHOP_CPU_PROFILE "Build a diagnostic binary with gperftools CPU profiling" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pyarrow; print(pyarrow.get_include()); print(pyarrow.get_library_dirs()[0])" OUTPUT_VARIABLE ARROW_PATHS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" ";" ARROW_PATHS "${ARROW_PATHS}")
list(GET ARROW_PATHS 0 ARROW_INCLUDE)
list(GET ARROW_PATHS 1 ARROW_LIB)
find_package(OpenMP REQUIRED)
add_executable(shopping src/main.cpp)
target_include_directories(shopping SYSTEM PRIVATE ${ARROW_INCLUDE})
target_compile_options(shopping PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
target_link_libraries(shopping PRIVATE OpenMP::OpenMP_CXX ${ARROW_LIB}/libparquet.so.2100 ${ARROW_LIB}/libarrow.so.2100)
if(SHOP_CPU_PROFILE)
  find_library(CPU_PROFILER profiler REQUIRED)
  target_link_libraries(shopping PRIVATE ${CPU_PROFILER})
  target_link_options(shopping PRIVATE -Wl,--no-as-needed)
  target_compile_options(shopping PRIVATE -g -fno-omit-frame-pointer)
endif()
set_target_properties(shopping PROPERTIES BUILD_RPATH "${ARROW_LIB}" INSTALL_RPATH "/opt/arrow")
install(TARGETS shopping DESTINATION bin)
add_executable(csv-check EXCLUDE_FROM_ALL tests/csv_check.cpp)
target_include_directories(csv-check PRIVATE src)
target_compile_options(csv-check PRIVATE -O2 -Wall -Wextra -Wpedantic)
add_executable(csv-scan-check EXCLUDE_FROM_ALL tests/csv_scan_check.cpp)
target_include_directories(csv-scan-check PRIVATE src)
target_compile_options(csv-scan-check PRIVATE -O2 -Wall -Wextra -Wpedantic)
